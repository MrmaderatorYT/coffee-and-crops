void run(Farmhand farmer) {
    final java.util.concurrent.atomic.AtomicInteger tally = new java.util.concurrent.atomic.AtomicInteger(0);
    farmer.workTogether(first -> {
        for (int i = 0; i < 3; i++) {
            first.harvest();
            tally.incrementAndGet();
            first.move(Direction.EAST);
        }
    }, second -> {
        for (int i = 0; i < 3; i++) {
            second.harvest();
            tally.incrementAndGet();
            second.move(Direction.EAST);
        }
    });
    assert tally.get() == 6 : "Expected one count for each crop";
}
