void run(Farmhand farmer) {
    java.util.HashMap<String, Integer> counts = new java.util.HashMap<>();
    for (int i = 0; i < 2; i++) {
        String crop = farmer.getCrop();
        counts.put(crop, counts.getOrDefault(crop, 0) + 1);
        farmer.harvest();
        farmer.move(Direction.EAST);
    }
    if (counts.getOrDefault("wheat", 0) == 2) {
        farmer.walkTo(3, 0);
    }
}
