void run(Farmhand farmer) {
    ArrayList<Integer> stops = new ArrayList<>();
    stops.add(1);
    stops.add(3);
    stops.add(5);
    java.util.Optional<Integer> stop = stops.stream().filter(cell -> cell >= 4).findFirst();
    int column = stop.orElse(0);
    farmer.walkTo(column, 0);
    farmer.harvest();
}
