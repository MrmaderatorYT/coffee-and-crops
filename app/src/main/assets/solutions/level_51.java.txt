void run(Farmhand farmer) {
    ArrayList<Integer> stops = new ArrayList<>();
    stops.add(1);
    stops.add(3);
    stops.add(5);
    ArrayList<Integer> route = new ArrayList<>();
    stops.stream().forEach(route::add);
    for (int cell : route) {
        farmer.walkTo(cell, 0);
        farmer.harvest();
    }
}
