void run(Farmhand farmer) {
    for (int i = 0; i < 3; i++) {
        switch (i) {
            case 0: {
                farmer.water();
            }
            case 1: {
                farmer.move(Direction.EAST);
                break;
            }
            default: {
                farmer.harvest();
                break;
            }
        }
    }
}
